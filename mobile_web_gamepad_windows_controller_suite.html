# =====================================================================
# AERO CONTROLLER - WINDOWS PC RECEIVER BRIDGE (pc_receiver.py)
# Emulates a Native Xbox 360 Controller via ViGEmBus driver & WebSockets
# =====================================================================
import asyncio
import json
import websockets
import vgamepad as vg

# Initialize Virtual Xbox 360 Gamepad Driver
gamepad = vg.VX360Gamepad()

async def handle_controller(websocket):
    print("\n[+] Mobile Web Controller Connected!")
    try:
        async for message in websocket:
            data = json.loads(message)
            
            # 1. Parse Gyroscope Motion Data (Roll -> Left Stick Steering)
            gyro = data.get('gyro', {})
            roll = gyro.get('roll', 0)
            pitch = gyro.get('pitch', 0)
            
            # 2. Parse Analog Stick Data
            left_stick = data.get('sticks', {}).get('left', {'x': 0, 'y': 0})
            right_stick = data.get('sticks', {}).get('right', {'x': 0, 'y': 0})
            
            # Apply Gyro Steering override if rolled significantly
            steer_x = max(-1.0, min(1.0, roll / 45.0)) if abs(roll) > 5 else left_stick['x']
            
            # Update Virtual Analog Joysticks
            gamepad.left_joystick_float(x_value_float=steer_x, y_value_float=-left_stick['y'])
            gamepad.right_joystick_float(x_value_float=right_stick['x'], y_value_float=-right_stick['y'])
            
            # 3. Map Button Inputs to Xbox Controller Buttons
            btns = data.get('buttons', {})
            
            # Action Buttons
            if btns.get('A'): gamepad.press_button(button=vg.XUSB_BUTTON.XUSB_GAMEPAD_A)
            else: gamepad.release_button(button=vg.XUSB_BUTTON.XUSB_GAMEPAD_A)
            
            if btns.get('B'): gamepad.press_button(button=vg.XUSB_BUTTON.XUSB_GAMEPAD_B)
            else: gamepad.release_button(button=vg.XUSB_BUTTON.XUSB_GAMEPAD_B)
            
            if btns.get('X'): gamepad.press_button(button=vg.XUSB_BUTTON.XUSB_GAMEPAD_X)
            else: gamepad.release_button(button=vg.XUSB_BUTTON.XUSB_GAMEPAD_X)
            
            if btns.get('Y'): gamepad.press_button(button=vg.XUSB_BUTTON.XUSB_GAMEPAD_Y)
            else: gamepad.release_button(button=vg.XUSB_BUTTON.XUSB_GAMEPAD_Y)
            
            # Bumpers & Menu
            if btns.get('L1'): gamepad.press_button(button=vg.XUSB_BUTTON.XUSB_GAMEPAD_LEFT_SHOULDER)
            else: gamepad.release_button(button=vg.XUSB_BUTTON.XUSB_GAMEPAD_LEFT_SHOULDER)
            
            if btns.get('R1'): gamepad.press_button(button=vg.XUSB_BUTTON.XUSB_GAMEPAD_RIGHT_SHOULDER)
            else: gamepad.release_button(button=vg.XUSB_BUTTON.XUSB_GAMEPAD_RIGHT_SHOULDER)
            
            if btns.get('start'): gamepad.press_button(button=vg.XUSB_BUTTON.XUSB_GAMEPAD_START)
            else: gamepad.release_button(button=vg.XUSB_BUTTON.XUSB_GAMEPAD_START)
            
            if btns.get('select'): gamepad.press_button(button=vg.XUSB_BUTTON.XUSB_GAMEPAD_BACK)
            else: gamepad.release_button(button=vg.XUSB_BUTTON.XUSB_GAMEPAD_BACK)

            # Analog Triggers (L2 / R2)
            gamepad.left_trigger_float(value_float=1.0 if btns.get('L2') else 0.0)
            gamepad.right_trigger_float(value_float=1.0 if btns.get('R2') else 0.0)
            
            # Update Virtual Hardware State
            gamepad.update()

    except websockets.exceptions.ConnectionClosed:
        print("[-] Mobile Controller Disconnected.")

async def main():
    print("=====================================================")
    print("  AERO PC GAMEPAD BRIDGE SERVER - LISTENING ON 8080  ")
    print("=====================================================")
    async with websockets.serve(handle_controller, "0.0.0.0", 8080):
        await asyncio.Future()

if __name__ == "__main__":
    asyncio.run(main())
